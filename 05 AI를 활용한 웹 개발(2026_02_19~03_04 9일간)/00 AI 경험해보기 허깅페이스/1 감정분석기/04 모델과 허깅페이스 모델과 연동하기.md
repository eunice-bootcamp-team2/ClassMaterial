```python
from django.db import models

class TextItem(models.Model):

    text = models.TextField()
    # 사용자가 입력한 원본 문장 저장
    # 길이 제한 없음 (긴 텍스트 가능)

    label = models.CharField(max_length=50, blank=True)
    # 감정 분석 결과 라벨 저장
    # 예: Positive / Negative / Neutral
    # blank=True → 값이 없어도 저장 가능 (초기에는 비어 있을 수 있음)

    score = models.FloatField(null=True, blank=True)
    # 감정 분석 신뢰도(확률) 저장
    # 예: 0.9876
    # null=True → DB에 NULL 허용
    # blank=True → 폼 검증 시 비어 있어도 허용

    created_at = models.DateTimeField(auto_now_add=True)
    # 레코드 최초 생성 시각 자동 저장
    # 수정해도 변경되지 않음
```

1. 분석 로직을 함수로 분리: `analyze_text()`

- UpdateView에서도 분석해야 하니까

CRUD 제너릭뷰 5개 추가
- ListView: 저장된 글 목록
- DetailView: 글 상세

- UpdateView: 글 수정(수정 시 다시 분석)
- DeleteView: 삭제

```python

# URL을 문자열 대신 이름으로 참조하기 위한 함수
# lazy → 실행 시점까지 평가 지연 (CBV에서 안전)

from django.views.generic import ListView, DetailView, CreateView, UpdateView, DeleteView
# Django의 제네릭 뷰(CBV)
# CRUD 패턴을 매우 간단하게 구현 가능

from .forms import SentimentForm
# 사용자 입력 폼 (text 입력용)

from .hf_model import get_classifier
# HuggingFace 모델 로딩 함수

from .models import TextItem
# DB 모델 (텍스트 + 감정 분석 결과 저장)

# ---------------------------------------------------------
# 1️⃣ 감정 분석 서비스 함수
# ---------------------------------------------------------
def analyze_text(text: str) -> dict:
    """
    입력된 텍스트를 HF 모델로 분석하여
    label / score 반환
    """

    # HuggingFace pipeline 객체 가져오기
    classifier = get_classifier()

    # 모델 추론 수행
    out = classifier(text)

    # 모델 출력 형식 안전 처리
    # pipeline은 보통 list 반환
    if isinstance(out, list) and out:
        out = out[0]

    # 필요한 값만 추출하여 반환
    return {
        "label": out.get("label") or "",           # 감정 라벨
        "score": float(out.get("score", 0.0)),     # 신뢰도 점수
    }


# ---------------------------------------------------------
# 2️⃣ 목록 화면 View
# ---------------------------------------------------------
class TextItemListView(ListView):

    model = TextItem
    # 어떤 모델 데이터를 사용할지 지정

    template_name = "sentiment/textitem_list.html"
    # 렌더링할 HTML 템플릿

    context_object_name = "items"
    # 템플릿에서 사용할 변수명

    ordering = ["-id"]
    # 최신 글이 위로 오도록 정렬

# 3️⃣ 상세 화면 View
# ---------------------------------------------------------
class TextItemDetailView(DetailView):

    model = TextItem
    template_name = "sentiment/textitem_detail.html"
    context_object_name = "item"
    # 단일 객체 전달


# ---------------------------------------------------------
# 4️⃣ 생성(Create) View
# ---------------------------------------------------------
class TextItemCreateView(CreateView):
    """
    글 작성 → 저장 시 자동 감정 분석
    """

    model = TextItem
    form_class = SentimentForm
    template_name = "sentiment/textitem_form.html"

    success_url = reverse_lazy("sentiment:item_list")
    # 저장 성공 시 이동할 URL

    def form_valid(self, form):

        text = form.cleaned_data["text"]

        # 감정 분석 수행
        analysis = analyze_text(text)

        # DB 저장 전에 분석 결과 주입
        form.instance.label = analysis["label"]
        form.instance.score = analysis["score"]

        # 기본 저장 로직 실행
        return super().form_valid(form)

    def get_success_url(self):
        # 수정 완료 후 상세 페이지로 이동
        return reverse_lazy(
            "sentiment:item_detail",
            kwargs={"pk": self.object.pk}
        )

    def form_valid(self, form):

        text = form.cleaned_data["text"]

        # 수정 시에도 재분석 수행
        analysis = analyze_text(text)

        form.instance.label = analysis["label"]
        form.instance.score = analysis["score"]

        return super().form_valid(form)

    model = TextItem

    success_url = reverse_lazy("sentiment:item_list")
    # 삭제 후 목록 이동
```

1️⃣ `analyze_text()` 함수 — 왜 따로 만들었을까?
```python
def analyze_text(text: str) -> dict:
    """
    HF 모델로 감정분석 후 label/score 반환
    """
```

🔍 이 함수의 역할
> 문장을 하나 받아서 → 허깅페이스 AI로 분석 → 결과를 정리해서 돌려주는 함수

쉽게 말해  
👉 AI에게 감정 분석만 시키는 전용 함수입니다.

---
실제 플랫폼에서는 AI 호출이 단순 함수 호출이 아니라 외부 서비스 호출에 가까운 작업입니다.

그래서 신경써야 하는 부분은:
- AI 결과 포맷이 바뀔 수 있음 (list/dict 등)
- 결과에 label/score가 없을 수 있음
- 모델 호출이 느릴 수 있음 (서비스 지연 원인)
- 향후 모델을 바꾸거나, API 서버로 분리할 수도 있음
그래서 AI 호출 코드는 한 군데로 모아두는 게 정석입니다.
`analyze_text()`는 AI 호출을 재사용 가능한 서비스 함수로 분리한 지점이라서 플랫폼 연동의 중심입니다.

---
처음(02 단계)에는 이런 코드가 `form_valid()` 안에 직접 있었습니다.
```python
classifier = get_classifier()
out = classifier(text)
```

그런데 04 단계가 되면:
- 새 글 작성할 때도 분석해야 하고
- 글 수정할 때도 다시 분석해야 합니다.
    
👉 그러면 같은 분석 코드가 여러 군데에서 반복됩니다.

공통함수빼내기:
> AI 분석은 하나의 독립된 작업이니까,  
> 따로 함수로 빼서 필요할 때마다 쓰는겁니다.

✅ 이렇게 분리하면 좋은 점
- 코드 중복 제거
- 나중에 분석 방식 바꿔도 여기만 수정하면 됨
- CreateView / UpdateView에서 같은 방식으로 재사용 가능

함수 내부
```python
classifier = get_classifier()
out = classifier(text)
```
- 이미 서버에 올라가 있는 허깅페이스 모델을 가져와서
- 사용자가 쓴 문장을 분석함

위의 코드를 한 줄로 합치면 사실 이렇게도 쓸 수 있습니다. 동작은 100% 같습니다.
```python
out = get_classifier()(text)
```
그럼에도 굳이 나눠 쓰는 이유는 역할이 완전히 다르기 때문입니다.

`classifier = get_classifier()` 의 진짜 의미는 감정 분석을 할 수 있는 AI 도구를 하나 꺼내온다 입니다.
- `get_classifier()`의 반환값은 함수처럼 호출 가능한 객체
- 즉, `classifier`는:
    - 문자열을 넣으면
    - 감정 분석 결과를 돌려주는 분석기 그 자체입니다.
비유하면:
```
classifier = 감정분석기
```
아직 분석은 안 했고, 기계만 준비해 둔 상태입니다.

`out = classifier(text)` 의 진짜 의미는 준비해 둔 감정분석기에 실제 문장을 넣어서 결과를 받아온다 입니다.

- `text` → 사용자가 입력한 순수 문자열
- `classifier(text)` →
    - 문자열을 AI에 넣고
    - AI가 계산을 끝낸 뒤
    - 결과를 반환
- 즉:
```
text        = 입력 재료
classifier  = 기계
out         = 기계가 만들어낸 결과물
```

그러면 왜 굳이 결과를 out이라는 변수에 옮겨 담을까요?
이유 ① 중간 결과를 눈에 보이게 만들기 위해서 입니다.
초보자든, 실무자든 디버깅의 80%는 중간값 확인입니다. 
```python
out = classifier(text)
print(out)
```
이 한 줄이 가능해지는 순간:
- AI가 어떤 구조로 응답하는지
- list인지 dict인지
- label/score가 있는지를 바로 확인할 수 있습니다.

만약 한 줄로 써버리면?
```python
return get_classifier()(text)[0]["label"]
```
- 어디서 에러 났는지
- 어떤 단계에서 값이 꼬였는지 확인하기 지옥이 되버립니다.

이유 ② 입력과 출력을 논리적으로 분리하기 위해서 입니다.

이건 플랫폼 설계에서 매우 중요해요.

|단계|의미|
|---|---|
|`text`|사용자 입력 (신뢰 ❌)|
|`classifier`|외부 시스템 (AI, 느림, 불안정)|
|`out`|외부 시스템의 응답 (형태 불안정)|

👉 그래서 AI의 응답은 반드시 한 번 변수에 받아서 검증/정리하는 습관을 들여야 합니다.

실제 서비스에서는:
```python
out = classifier(text)

if out is None:
    ...
if isinstance(out, list):
    ...
if "label" not in out:
    ...
```
이런 방어 코드가 계속 붙습니다.

이유 ③ AI 호출은 계산이 아니라 사건(event)이기 때문입니다.

이게 제일 중요한 개념이야.
```python
classifier(text)
```

이건 단순한 계산이 아니라:
- 시간이 걸릴 수 있고
- 실패할 수 있고
- 결과 형태가 바뀔 수 있는 외부 이벤트 호출입니다.
    
그래서 실무에서는 거의 항상:
```python
result = external_call(input)
```
처럼 반드시 변수에 담습니다.
DB 쿼리도, API 호출도, AI 호출도 다 똑같습니다.

```python
if isinstance(out, list) and out:
    out = out[0]
```
- HF 결과는 보통 리스트로 오기 때문에
- 첫 번째 결과만 꺼내서 사용

### 02 → 04 단계 전환에 따라 forms.py가 변경되는 이유

02 단계의 Form
```python
class SentimentForm(forms.Form):
    text = forms.CharField(...)
```

왜 이렇게 했냐?
- DB가 없으니까 입력만 받는 폼이면 충분했습니다.

04 단계의 ModelForm
`sentiment/forms.py`
```python
from django import forms
from .models import TextItem

class SentimentForm(forms.ModelForm):
    class Meta:
        model = TextItem
        fields = ["text"]
```

왜 이렇게 바뀌냐?
- 이제는 글을 DB에 저장해야 하므로,
- 폼이 단순 입력 도구가 아니라 TextItem을 만들기 위한 폼이 되어야 합니다.
