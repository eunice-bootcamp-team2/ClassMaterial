    print(f"[{name}] p-value       :", result[1])
    print("-" * 40)
    return result


# ---------------------------------------------------------
# (1) 원본 시계열 준비

# ts = df2["시가"] 처럼 “값 컬럼”만 뽑아서 시계열로 둔다

# 원본 정상성 확인(선택)
res_original = check_stationarity(ts, "original")

# ---------------------------------------------------------
ts_diff1 = ts.diff()               # 현재값 - 이전값

# ---------------------------------------------------------
# (2-1) 1차 차분 결과 시각화 (선택)
# ---------------------------------------------------------
px.line(df2, x="Date", y="diff1")  # (선택) 시각화

# ---------------------------------------------------------
# (2-2) 1차 차분 정상성 검정(ADF)
# ---------------------------------------------------------
res_diff1 = check_stationarity(ts_diff1, "diff1")



# ---------------------------------------------------------
# (3) 2차 차분 (1차로 부족하면)
# ---------------------------------------------------------

ts_diff2 = ts.diff(2)

# 방법 B: 1차 차분을 한 번 더 차분(= 누적 차분 느낌)
# ts_diff2 = ts_diff1.diff()

df2["diff2"] = ts_diff2

# ---------------------------------------------------------
# (3-1) 2차 차분 결과 시각화 (선택)
# ---------------------------------------------------------
px.line(df2, x="Date", y="diff2")

# ---------------------------------------------------------
# (3-2) 2차 차분 정상성 검정(ADF)
# ---------------------------------------------------------
res_diff2 = check_stationarity(ts_diff2, "diff2")
res_diff2 = check_stationarity(ts_diff2, "diff2")


# ---------------------------------------------------------
# (4) 계절 차분 (계절성이 원인이라면)
# ---------------------------------------------------------
# 주기 s가 있을 때 (예: 7일, 12개월 등)
# ts_seasonal_diff = ts.diff(s)
# check_stationarity(ts_seasonal_diff, f"seasonal_diff_{s}")


# ---------------------------------------------------------
# (5) 최종 판단 규칙
# ---------------------------------------------------------
# - p-value <= 0.05  → 정상성(O) → ARIMA 등 적용 가능
# - p-value >  0.05  → 정상성(X) → 다음 변환으로 이동
#
# 실무 팁:
# - 보통 1차(diff1)에서 끝나는 경우가 많음(특히 주가)
# - 2차(diff2)까지 가면 과차분 위험도 있으니 그래프/ACF도 같이 보는 편

차분(Differencing) + 정상성 재검정
```python
# =========================================================
# 07 차분(Differencing)으로 정상성(ADF) 만들기 (Skeleton 패턴 순서 그대로)
# (0) 준비 → (0-1) ADF 출력 헬퍼 → (1) 원본 시계열 →
# (2) 1차 차분 → (2-1) 시각화 → (2-2) ADF →
# (3) 2차 차분(필요 시) → (3-1) 시각화 → (3-2) ADF → (4) 판단
# =========================================================

# ---------------------------------------------------------
# (0) 준비: 라이브러리 불러오기
# ---------------------------------------------------------
import plotly.express as px
from statsmodels.tsa.stattools import adfuller


# ---------------------------------------------------------
# (0-1) ADF(정상성) 결과를 보기 좋게 출력하는 헬퍼 함수
# ---------------------------------------------------------
def check_stationarity(series, name="data"):
    result = adfuller(series.dropna())
    print(f"[{name}] ADF Statistic :", result[0])
    print(f"[{name}] p-value       :", result[1])
    print("-" * 40)
    return result



# ---------------------------------------------------------
# (1) 원본 시계열 준비 (+ 원본 정상성 확인: 선택)
# ---------------------------------------------------------
# 전제: df2에는 최소한 아래 컬럼이 있어야 함
# - df2["Date"] : 날짜(datetime)
# - df2["시가"] : 값(주가 시가)
ts = df2["시가"]

# (선택) 원본이 정상성인지 먼저 확인
res_original = check_stationarity(ts, "original")


# ---------------------------------------------------------
# (2) 1차 차분 수행 (기본 카드)
# ---------------------------------------------------------
# diff() : 현재 값 - 이전 값
# 예) 6/12 시가 - 6/11 시가
df2["시가_diff"] = ts.diff()
ts_diff1 = df2["시가_diff"]

# ---------------------------------------------------------
# (2-1) 1차 차분 결과 시각화 (선택)
# ---------------------------------------------------------
px.line(df2, x="Date", y="시가_diff", title="1st Difference")


# ---------------------------------------------------------
# (2-2) 1차 차분 정상성 검정(ADF)
# ---------------------------------------------------------
res_diff1 = check_stationarity(ts_diff1, "diff1")
# 해석:
# p-value <= 0.05  → 정상성(O)
# p-value  > 0.05  → 정상성(X) → 2차 차분 등 추가 시도




# ---------------------------------------------------------
# (3) 2차 차분 (1차로 부족할 때만)
# ---------------------------------------------------------
# diff(2) : 현재 값 - 2시점 이전 값
df2["시가_diff2"] = ts.diff(2)
ts_diff2 = df2["시가_diff2"]

# ---------------------------------------------------------
# (3-1) 2차 차분 결과 시각화 (선택)
# ---------------------------------------------------------**
px.line(df2, x="Date", y="시가_diff2", title="2nd Difference")

# ---------------------------------------------------------
# (3-2) 2차 차분 정상성 검정(ADF)
# ---------------------------------------------------------
res_diff2 = check_stationarity(ts_diff2, "diff2")



# ---------------------------------------------------------
# (4) 판단(해석 기준)
# ---------------------------------------------------------
# - p-value <= 0.05 : 정상성 통과 → ARIMA 같은 시계열 모델 적용 가능
# - p-value >  0.05 : 여전히 비정상 → 계절 차분/로그+차분/수익률 등 다음 단계 고려
```

0️⃣ 이 단계의 위치

우리는 삼성 주가 시계열에 대해 다음을 시도했다:

1. 로그 변환 / Box-Cox 변환 ❌
2. 계절성 제거
    - Classical Decomposition ❌
    - STL Decomposition ❌

👉 여전히 ADF Test에서 정상성 실패

그래서 등장하는 가장 강력하고 실무에서 가장 많이 쓰이는 방법이 바로

➡️ 차분(Differencing) 입니다.

---
1️⃣ 차분이란 무엇인가?

한 줄 정의
> 차분 = 현재 시점 값 − 이전 시점 값

- 영어: _Lag Difference_
- 값이 아니라 변화량을 분석 대상으로 삼는 것

###### 직관적인 예시
| 날짜  | 시가     |
| --- | ------ |
| 어제  | 59,000 |
| 오늘  | 60,000 |

➡️ 차분값 = +1,000

즉,
❌ 오늘 주가가 얼마인가
⭕ 어제 대비 얼마나 변했는가

---
2️⃣ 왜 차분을 하면 정상성이 되는가?

차분을 적용하면 다음 효과가 있다:

① 추세 제거
- 장기 우상향 / 우하향 흐름 제거
- 계속 올라가는 주가 → 위아래로 흔들리는 변화량

② 평균 안정화
- 평균이 0 근처로 수렴
- 시계열이 중심을 기준으로 움직임

③ 분산 안정화
- 시간에 따라 커지던 변동 폭이 줄어듦
👉 비정상 시계열의 핵심 원인(추세, 스케일 증가)을 한 번에 제거



---
3️⃣ 1차 차분 (First Differencing)

개념
> 1차 차분 = 바로 이전 시점과의 차이

코드
```python
df2['시가_diff'] = df2['시가'].diff()
```

특징
- 첫 번째 값은 NaN
    - 이전 값이 없기 때문
- 이후 값은:
    - (오늘 − 어제)
    - (내일 − 오늘)

➡️ 이것을 1차 차분이라고 부른다

---
4️⃣ 차분 결과 시각화 해석

차분된 그래프를 보면:
- 장기 추세가 사라짐
- 평균이 0 근처
- 변동 폭이 비교적 일정
- “가격 그래프”가 아니라 **“변화량 그래프”처럼 보임**

핵심:
> 주가는 어제 가격을 기준으로 움직이기 때문에
> 차분하면 정상성을 만족하는 경우가 굉장히 많다.

---
5️⃣ 차분 후 정상성 검정 (ADF Test)

⚠️ 반드시 주의할 점
- `diff()` 결과에는 NaN 포함
- ADF Test 전에 dropna() 필수
```python
from statsmodels.tsa.stattoolsimport adfuller

adfuller(df2['시가_diff'].dropna())
```

결과 해석
- p-value < 0.05
- 귀무가설 기각
- 👉 차분된 시계열은 정상성을 만족

즉,

| 데이터   | 정상성 |
| ----- | --- |
| 원본 시가 | ❌   |
| 차분 시가 | ⭕   |

---
6️⃣ 사고방식의 전환 (아주 중요 ⭐)
❌ 주가 자체를 예측한다
⭕ 주가의 변화량을 예측한다

실제 예측 흐름
1. 모델이 차분값(변화량) 예측
2. 마지막 실제 주가에 더함
3. 다음 시점 주가 복원
4. 반복 → n일 후 예측

👉 ARIMA, SARIMA의 핵심 아이디어

---
7️⃣ 2차 차분 (Second Differencing)

언제 필요한가?
- 1차 차분 후에도:
    - 추세가 남아 있거나
    - 정상성 실패 시

개념
> 2차 차분 = 1차 차분 결과를 다시 차분

코드
```python
df2['시가_diff2'] = df2['시가'].diff(2)
```

또는
```python
df2['시가'].diff().diff()
```

주의사항 ⚠️
- 차분을 많이 할수록 정보 손실
- 1차 차분으로 충분하면 거기서 멈춘다
- ARIMA의 `d` 값은 보통 0~2

---
8️⃣ 이 파트의 최종 결론
- 로그 변환 ❌
- 계절성 제거 ❌
- 차분 ⭕ (성공)

👉 드디어 정상성을 만족하는 시계열 확보

이제 다음 단계는 자연스럽게:

> 정상 시계열을 전제로 하는 모델 → AR / MA / ARIMA

---
9️⃣ 한 문장 요약

> 차분은 시계열 데이터의 시차 차이를 이용해 추세를 제거하고 변동성을 안정화하여 정상성을 만족시키는 가장 대표적인 기법이다.







## 0) 이 파트의 위치 (맥락)

앞에서 우리는 이미:

- 로그 변환 / Box-Cox 변환 ❌
    
- 계절성 제거 (Classical, STL) ❌
    

👉 **여전히 정상성 실패**

그래서 **마지막으로 가장 강력하고 자주 쓰는 방법**인  
➡️ **차분(Differencing)** 을 적용한다.

---

# 1) 차분이란 무엇인가?

## 정의 (아주 간단)

> **차분 = 현재 시점 값 − 이전 시점 값**

- 영어로는 **Lag Difference(시차 차이)**
    
- “차이”를 계산하는 것
    

### 예시

- 오늘 주가: 60,000
    
- 어제 주가: 59,000  
    ➡️ 차분값 = **+1,000**
    

이 “변화량” 자체를 새로운 시계열 데이터로 사용한다.

---

# 2) 왜 차분을 쓰는가?

차분을 하면 다음 효과가 있다:

1. **추세 제거**
    
    - 장기적으로 우상향/우하향하던 흐름이 사라짐
        
2. **변동성 안정화**
    
    - 평균이 0 근처로 수렴
        
    - 값의 스케일이 일정해짐
        
3. **정상성 만족 가능성 ↑**
    
    - 실제로 많은 비정상 시계열이  
        👉 차분 한 번으로 정상성이 됨
        

그래서 **시계열 분석에서 차분은 거의 필수**처럼 쓰인다.

---

# 3) 1차 차분 (First Difference)

## 개념

> **1차 차분 = 바로 이전 값과의 차이**

### pandas에서

`df2['시가_diff'] = df2['시가'].diff()`

### 이때 생기는 현상

- 맨 첫 값은 **NaN**
    
    - 이전 값이 없어서 차이를 계산할 수 없기 때문
        
- 이후 값들은:
    
    - (오늘 − 어제), (내일 − 오늘) … 이런 식으로 채워짐
        

👉 이것을 **1차 차분**이라고 부른다.

---

# 4) 차분 결과 시각화

차분된 값을 그래프로 보면:

- 평균이 **0 근처**
    
- 큰 추세가 사라짐
    
- 변동 폭이 비교적 일정
    
- 중간중간 이상치는 있지만, 폭발적으로 커지지 않음
    

강사의 핵심 설명:

> “주가는 어제 가격을 기준으로 움직이기 때문에  
> 차분을 하면 대부분 정상성을 만족한다.”

---

# 5) 차분 후 정상성 검정 (ADF Test)

## 주의사항 (중요)

- 차분 결과에는 **NaN(결측값)** 이 있음
    
- ADF Test 전에 **NaN 제거 필수**
    

### 결과

- p-value < 0.05
    
- 귀무가설 기각
    
- 👉 **차분된 시계열은 정상성을 만족**
    

즉,

> **원본 데이터는 비정상**  
> **차분된 데이터는 정상**

---

# 6) 여기서 사고방식이 바뀐다 (아주 중요)

❌ “주가 자체를 예측한다”  
⭕ **“주가의 변화량(차분)을 예측한다”**

### 왜?

- 차분은 정상성을 만족 → 모델 적용 가능
    
- 모델이 예측하는 건:
    
    - “내일 가격” ❌
        
    - **“내일 얼마나 변할지” ⭕**
        

### 예측 흐름

1. 차분값 예측
    
2. 마지막 실제 값에 **차분을 더함**
    
3. 다음 날 값 계산
    
4. 이를 반복해서 n일 후까지 예측
    

---

# 7) 2차 차분 (Second Difference)

## 언제 쓰는가?

- 1차 차분을 했는데도:
    
    - 여전히 추세/패턴이 남아 있으면
        

## 개념

> **2차 차분 = 1차 차분 결과를 다시 차분**

### pandas에서

`df2['시가_diff2'] = df2['시가'].diff(2)`

또는

`df2['시가'].diff().diff()`

### 특징

- 값의 스케일이 더 줄어들 수 있음
    
- 하지만:
    
    - 부호(+, −) 때문에 값이 더 커 보일 수도 있음
        

⚠️ **차분을 많이 한다고 좋은 건 아님**

- 필요 이상으로 하면 정보 손실
    
- 👉 **1차 차분이 정상성을 만족하면 거기서 멈춤**
