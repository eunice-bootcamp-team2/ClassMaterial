version: "3.9"

services:
  db:
    image: ankane/pgvector:v0.5.1
    container_name: product_review_postgres
    restart: always

    # [개발용] 개발 환경변수 파일 사용

      - .env.dev

    networks:
      - app-network

  redis:
    image: redis:7
    container_name: redis-server
    restart: always

      - "6379:6379"

    networks:
      - app-network

  web:
    build: .
    container_name: drf-web

    # [개발용] Django 개발 서버 사용
    command: python manage.py runserver 0.0.0.0:8000

    restart: always

    # [개발용] 코드 변경 즉시 반영되도록 볼륨 마운트

      - "8000:8000"

      - .env.dev

      - db
      - redis
      - fastapi

    networks:
      - app-network

  celery:
    build: .
    container_name: celery-worker

    # [개발용] Celery worker 실행

    restart: always

    # [개발용] 코드 변경 즉시 반영
    volumes:
      - .:/app

    # [개발용] .env.dev 사용

      - .env.dev

from models.embedding_model import embedding_model
from sklearn.metrics.pairwise import cosine_similarity


def make_embeddings(texts: list[str]) -> list[list[float]]:
    """
    여러 문장을 받아 임베딩 벡터 리스트로 반환
    """
    vectors = embedding_model.encode(texts)
    return [vector.tolist() for vector in vectors]


def calculate_similarity(text1: str, text2: str) -> float:
    """
    두 문장의 cosine similarity 계산
    """
    vectors = embedding_model.encode([text1, text2])
    score = cosine_similarity([vectors[0]], [vectors[1]])[0][0]
    return float(score)
