### 1️⃣ 예외 처리 – 공통 예외 클래스 + 전역 핸들러

1-1. 도메인 예외 정의

📁 `app/core/exceptions.py`
```python
from fastapi import status

class AppException(Exception):
    """모든 커스텀 예외의 부모 클래스"""

    status_code: int = status.HTTP_400_BAD_REQUEST
    error_code: str = "APP_ERROR"
    message: str = "Application error"

    def __init__(self, message: str | None = None):
        if message:
            self.message = message
        super().__init__(self.message)

class NotFoundException(AppException):
    status_code = status.HTTP_404_NOT_FOUND
    error_code = "NOT_FOUND"
    message = "Resource not found"

class PermissionDeniedException(AppException):
    status_code = status.HTTP_403_FORBIDDEN
    error_code = "PERMISSION_DENIED"
    message = "Access denied"

class ValidationException(AppException):
    status_code = status.HTTP_422_UNPROCESSABLE_ENTITY
    error_code = "VALIDATION_ERROR"
    message = "Invalid input"
```

> 앞으로 서비스 코드에서 raise NotFoundException("Todo not found") 처럼
> **비즈니스 로직에서 의미 있는 예외만 던지고**,
> “JSON으로 어떻게 보여줄지”는 전역 핸들러가 처리하게 할 거예요.

1-2. 전역 예외 핸들러 등록

📁 `app/core/error_handlers.py`
```python
from fastapi import FastAPI, Request
from fastapi.responses import JSONResponse
from fastapi.exceptions import RequestValidationError
from starlette import status

from app.core.exceptions import AppException

def register_exception_handlers(app: FastAPI) -> None:
    """FastAPI 앱에 전역 예외 핸들러들을 등록"""

    @app.exception_handler(AppException)
    async def app_exception_handler(request: Request, exc: AppException):
        return JSONResponse(
            status_code=exc.status_code,
            content={
                "error": {
                    "code": exc.error_code,
                    "message": exc.message,
                }
            },
        )

    @app.exception_handler(RequestValidationError)
    async def validation_exception_handler(request: Request, exc: RequestValidationError):
        # Pydantic/Request body, query, path 검증 실패
        return JSONResponse(
            status_code=status.HTTP_422_UNPROCESSABLE_ENTITY,
            content={
                "error": {
                    "code": "REQUEST_VALIDATION_ERROR",
                    "message": "Invalid request payload",
                    "details": exc.errors(),
                }
            },
        )

    @app.exception_handler(Exception)
    async def global_exception_handler(request: Request, exc: Exception):
        # 알 수 없는 서버 에러 (로그는 별도에 남기고, 사용자에게는 일반 메시지만)
        return JSONResponse(
            status_code=status.HTTP_500_INTERNAL_SERVER_ERROR,
            content={
                "error": {
                    "code": "INTERNAL_SERVER_ERROR",
                    "message": "Unexpected server error",
                }
            },
        )
```

### 2️⃣ 미들웨어 – 요청 시간 / Request-ID 로깅

2-1. 미들웨어 구현

📁 `app/core/middleware.py`
```python
import time
import uuid
from typing import Callable

from fastapi import Request, Response
from starlette.middleware.base import BaseHTTPMiddleware

class RequestLoggingMiddleware(BaseHTTPMiddleware):
    """
    - 각 요청마다 Request-ID 헤더 추가
    - 처리 시간(ms) 로깅 (지금은 print, 나중에 logger 로 변경 가능)
    """

    async def dispatch(self, request: Request, call_next: Callable) -> Response:
        request_id = str(uuid.uuid4())
        start_time = time.time()

        # 요청 context에 request_id 저장 (필요 시)
        request.state.request_id = request_id

        response: Response = await call_next(request)

        process_time = (time.time() - start_time) * 1000  # ms
        response.headers["X-Request-ID"] = request_id
        response.headers["X-Process-Time-ms"] = f"{process_time:.2f}"

        # 여기는 실제 서비스에서는 logging 라이브러리 사용 권장
        print(
            f"[Request] {request.method} {request.url.path} "
            f"status={response.status_code} "
            f"request_id={request_id} "
            f"time_ms={process_time:.2f}"
        )

        return response
```

### 3️⃣ 파일 업로드 확장 – 크기 / MIME 타입 검증

앞에서 **Todo 첨부파일 업로드 API**를 만들었다고 했으니,
그 엔드포인트에 **검증 로직을 강화**해볼게요.

예:

📁 `app/api/routes/attachments.py` 또는 `todos.py` 안의 첨부 업로드 부분
```python
from fastapi import APIRouter, Depends, File, UploadFile, HTTPException, status
from sqlalchemy.orm import Session

from app.core.db import get_db
from app.core.security import get_current_user
from app.schemas.users import UserRead
from app.services.attachments_service import save_attachment_for_todo

router = APIRouter()

MAX_FILE_SIZE_MB = 5
ALLOWED_CONTENT_TYPES = {"image/jpeg", "image/png", "application/pdf"}

@router.post(
    "/{todo_id}/attachments",
    summary="Todo 첨부 파일 업로드",
)
async def upload_todo_attachment(
    todo_id: int,
    file: UploadFile = File(...),
    current_user: UserRead = Depends(get_current_user),
    db: Session = Depends(get_db),
):
    # 1) MIME 타입 검증
    if file.content_type not in ALLOWED_CONTENT_TYPES:
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST,
            detail=f"Unsupported file type: {file.content_type}",
        )

    # 2) 파일 사이즈 검증 (스트림으로 한 번 읽되, 너무 크면 중단)
    contents = await file.read()

    file_size_mb = len(contents) / (1024 * 1024)
    if file_size_mb > MAX_FILE_SIZE_MB:
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST,
            detail=f"File too large ({file_size_mb:.2f} MB). "
            f"Max allowed is {MAX_FILE_SIZE_MB} MB",
        )

    # 3) 실제 저장 로직 (서비스 레이어 호출)
    attachment = save_attachment_for_todo(
        db=db,
        user=current_user,
        todo_id=todo_id,
        filename=file.filename,
        content_type=file.content_type,
        data=contents,
    )

    return {
        "id": attachment.id,
        "filename": attachment.original_filename,
        "content_type": attachment.content_type,
        "size_mb": round(file_size_mb, 2),
    }
```

> 여기서 save_attachment_for_todo 는
> 
> **DB에 메타데이터 저장 + 실제 파일은 디스크/S3 경로에 저장**하는 함수라고 가정.

이제 Swagger `/docs` 에서도

- ■ 업로드 필드가 표시되고
- ■ 잘못된 파일 타입/크기일 때 예외가 **일관된 JSON**으로 떨어집니다.

📁 `tests/test_auth_and_todos.py`
```python
def signup_and_login(client):
    # 1) 회원가입
    resp = client.post(
        "/api/auth/signup",
        json={
            "username": "testuser",
            "email": "test@example.com",
            "password": "test1234",
        },
    )
    assert resp.status_code == 201
    user = resp.json()
    assert user["username"] == "testuser"

    # 2) 로그인
    resp = client.post(
        "/api/auth/login",
        data={
            "username": "testuser",
            "password": "test1234",
        },
        headers={"Content-Type": "application/x-www-form-urlencoded"},
    )
    assert resp.status_code == 200
    token_data = resp.json()
    assert "access_token" in token_data
    return token_data["access_token"]

def test_todo_crud_flow(client):
    access_token = signup_and_login(client)

    headers = {"Authorization": f"Bearer {access_token}"}

    # 1) 빈 목록 조회
    resp = client.get("/api/todos/", headers=headers)
    assert resp.status_code == 200
    assert resp.json() == []

    # 2) Todo 생성
    resp = client.post(
        "/api/todos/",
        headers=headers,
        json={
            "title": "First Task",
            "description": "Testing todo creation",
            "status": "todo",
            "priority": 3,
        },
    )
    assert resp.status_code == 201
    todo = resp.json()
    assert todo["title"] == "First Task"
    todo_id = todo["id"]

    # 3) 상세 조회
    resp = client.get(f"/api/todos/{todo_id}", headers=headers)
    assert resp.status_code == 200
    detail = resp.json()
    assert detail["id"] == todo_id
    assert detail["title"] == "First Task"

    # 4) 검색/필터
    resp = client.get("/api/todos/?status_filter=todo&q=First", headers=headers)
    assert resp.status_code == 200
    items = resp.json()
    assert len(items) == 1
    assert items[0]["id"] == todo_id
```
