`app/templates/index.html`
```html
<!DOCTYPE html>
<html lang="ko">

<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">

    <!-- 브라우저 탭 제목 -->
    <title>FastAPI Todo</title>

    <!-- 정적 CSS 파일 연결 -->
    <link rel="stylesheet" href="/static/style.css">
</head>

<body>

    <div class="todo-container">

        <!-- 페이지 제목 -->
        <h1>FastAPI Todo</h1>

        <!-- =========================
             로그인 / 로그아웃 UI 영역
             - 로그인 전: 아이디/비밀번호 입력창 표시
             - 로그인 후: 사용자 정보와 로그아웃 버튼 표시
        ========================== -->
        <div id="auth-box">

            <!-- 로그인 폼 -->
            <div id="login-form">
                <input type="text" id="login-username" placeholder="아이디">
                <input type="password" id="login-password" placeholder="비밀번호">
                <button onclick="login()">로그인</button>
                <button onclick="signup()">회원가입</button>
            </div>

            <!-- 로그인 후 표시되는 사용자 정보 -->
            <div id="user-info" style="display:none;">
                <span id="welcome-text"></span>
                <button onclick="logout()">로그아웃</button>
            </div>

        </div>

        <hr>

        <!-- =========================
             Todo 생성 영역
             - 입력창에 할 일 제목 입력
             - 추가 버튼 클릭 시 createTodo() 실행
        ========================== -->
        <div class="todo-input-box">
            <input type="text" id="create-title" placeholder="할 일을 입력하세요">
            <button onclick="createTodo()">추가</button>
        </div>

        <!-- 성공 / 실패 메시지를 보여주는 영역 -->
        <div id="message-box"></div>

        <!-- Todo 목록이 렌더링되는 영역 -->
        <ul id="todo-list" class="todo-list"></ul>

    </div>

<script>

let token = localStorage.getItem("token");
// =========================
// 브라우저 localStorage에 저장된 JWT 토큰을 꺼내옵니다.
// - 새로고침 후에도 로그인 상태를 유지하기 위해 사용
// - 로그인 성공 시 token을 저장하고
// - 이후 Todo API 호출 시 Authorization 헤더에 사용합니다.
// =========================


function showMessage(message, isError = false) {
    // =========================
    // 사용자에게 성공/실패 메시지를 화면에 보여주는 함수
    // - isError가 true이면 error 스타일
    // - false이면 success 스타일
    // =========================
    const box = document.getElementById("message-box");
    box.innerHTML = `<p class="${isError ? 'error' : 'success'}">${message}</p>`;
}


function setLoggedIn(username) {
    // =========================
    // 로그인 성공 후 UI를 변경하는 함수
    // - 로그인 폼은 숨기고
    // - 사용자 정보 영역을 보여줍니다.
    // - 상단에 "OOO님 로그인" 문구를 출력합니다.
    // =========================
    document.getElementById("login-form").style.display = "none";
    document.getElementById("user-info").style.display = "block";
    document.getElementById("welcome-text").innerText = `${username}님 로그인`;
}


function setLoggedOut() {
    // =========================
    // 로그아웃 상태의 UI로 되돌리는 함수
    // - 로그인 폼을 다시 보이게 하고
    // - 사용자 정보 영역은 숨깁니다.
    // =========================
    document.getElementById("login-form").style.display = "block";
    document.getElementById("user-info").style.display = "none";
}


async function signup() {

    // =========================
    // 회원가입 요청 함수
    // - 입력한 아이디/비밀번호를 읽어서
    // - FastAPI의 /users/signup API로 POST 요청을 보냅니다.
    // - 성공 시 회원가입 성공 메시지 출력
    // - 실패 시 서버의 detail 메시지 출력
    // =========================

    const username = document.getElementById("login-username").value;
    const password = document.getElementById("login-password").value;

    const res = await fetch("/users/signup", {
        method: "POST",
        headers: {
            "Content-Type": "application/json"
        },
        body: JSON.stringify({
            username: username,
            password: password
        })
    });

    const data = await res.json();

    if (!res.ok) {
        showMessage(data.detail || "회원가입 실패", true);
        return;
    }

    showMessage("회원가입 성공");
}


async function login() {

    // =========================
    // 로그인 요청 함수
    // - 입력한 아이디/비밀번호를 /users/login API로 전송
    // - 서버가 JWT access_token을 발급하면
    //   1) token 변수에 저장
    //   2) localStorage에도 저장
    // - 이후 로그인 UI 전환, Todo 목록 조회
    // =========================

    const username = document.getElementById("login-username").value;
    const password = document.getElementById("login-password").value;

    const res = await fetch("/users/login", {
        method: "POST",
        headers: {
            "Content-Type": "application/json"
        },
        body: JSON.stringify({
            username: username,
            password: password
        })
    });

    const data = await res.json();

    if (!res.ok) {
        showMessage(data.detail || "로그인 실패", true);
        return;
    }

    // 서버가 발급한 JWT 토큰 저장
    token = data.access_token;
    localStorage.setItem("token", token);

    // 로그인 UI로 전환
    setLoggedIn(username);

    // 로그인한 사용자의 Todo 목록 조회
    loadTodos();

    showMessage("로그인 성공");
}


function logout() {

    // =========================
    // 로그아웃 처리 함수
    // - localStorage에 저장된 JWT 토큰 삭제
    // - 현재 메모리의 token도 null 처리
    // - UI를 로그아웃 상태로 전환
    // - Todo 목록 화면도 비웁니다.
    // =========================

    localStorage.removeItem("token");

    token = null;

    setLoggedOut();

    document.getElementById("todo-list").innerHTML = "";

    showMessage("로그아웃 되었습니다.");
}


async function loadTodos() {

    // =========================
    // Todo 목록 조회 함수
    // - 로그인된 상태에서만 실행
    // - /todos API에 GET 요청
    // - Authorization 헤더에 Bearer 토큰 포함
    // - 서버가 돌려준 Todo 목록을 화면에 렌더링
    // =========================

    if (!token) {
        return;
    }

    const res = await fetch("/todos", {
        headers: {
            "Authorization": "Bearer " + token
        }
    });

    const todos = await res.json();

    const list = document.getElementById("todo-list");

    list.innerHTML = "";

    todos.forEach(todo => {

        const li = document.createElement("li");

        li.className = "todo-item";

        li.innerHTML = `
        <span class="${todo.done ? 'completed' : ''}">
            ${todo.title}
        </span>

        <div>
            <button onclick="toggleDone(${todo.id}, ${todo.done})">완료</button>
            <button onclick="deleteTodo(${todo.id})">삭제</button>
        </div>
        `;

        list.appendChild(li);

    });

}


async function createTodo() {

    // =========================
    // Todo 생성 함수
    // - 입력창의 제목 값을 읽어서 /todos API로 POST 요청
    // - JWT 토큰을 Authorization 헤더에 포함
    // - 성공하면 입력창 비우고 목록 새로고침
    // =========================

    const title = document.getElementById("create-title").value;

    if (!title) {
        showMessage("할 일을 입력하세요", true);
        return;
    }

    const res = await fetch("/todos", {
        method: "POST",
        headers: {
            "Content-Type": "application/json",
            "Authorization": "Bearer " + token
        },
        body: JSON.stringify({
            title: title,
            done: false
        })
    });

    const data = await res.json();

    if (!res.ok) {
        showMessage(data.detail || "생성 실패", true);
        return;
    }

    showMessage("Todo 생성 완료");

    document.getElementById("create-title").value = "";

    loadTodos();
}


async function deleteTodo(id) {

    // =========================
    // Todo 삭제 함수
    // - 삭제할 Todo의 id를 받아서 /todos/{id} API로 DELETE 요청
    // - JWT 토큰을 함께 보내서 인증된 사용자만 삭제 가능
    // - 성공하면 목록 새로고침
    // =========================

    const res = await fetch(`/todos/${id}`, {
        method: "DELETE",
        headers: {
            "Authorization": "Bearer " + token
        }
    });

    if (!res.ok) {
        showMessage("삭제 실패", true);
        return;
    }

    showMessage("삭제 완료");

    loadTodos();
}


async function toggleDone(id, done) {

    // =========================
    // Todo 완료 상태 변경 함수
    // - 현재 done 값을 반대로 바꿔서 /todos/{id} API로 PUT 요청
    // - 예:
    //   false -> true
    //   true -> false
    // - 성공하면 목록 새로고침
    // =========================

    const res = await fetch(`/todos/${id}`, {
        method: "PUT",
        headers: {
            "Content-Type": "application/json",
            "Authorization": "Bearer " + token
        },
        body: JSON.stringify({
            done: !done
        })
    });

    if (!res.ok) {
        showMessage("수정 실패", true);
        return;
    }

    loadTodos();
}


if (token) {

    // =========================
    // 페이지가 처음 열릴 때 자동 로그인 상태 확인
    // - localStorage에 token이 있으면 /users/me API 호출
    // - 토큰이 유효하면:
    //   1) 사용자 이름 표시
    //   2) Todo 목록 자동 조회
    // - 토큰이 만료되었거나 잘못되었으면 logout() 처리
    // =========================

    fetch("/users/me", {
        headers: {
            "Authorization": "Bearer " + token
        }
    })
    .then(res => res.json())
    .then(data => {
        setLoggedIn(data.username);
        loadTodos();
    })
    .catch(() => {
        logout();
    });

}

</script>

</body>
</html>
```
