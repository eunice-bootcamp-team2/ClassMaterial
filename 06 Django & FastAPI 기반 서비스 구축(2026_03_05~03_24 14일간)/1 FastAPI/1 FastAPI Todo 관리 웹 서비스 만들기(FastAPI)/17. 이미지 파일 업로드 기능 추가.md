```python
# app/utils/file_handler.py

import os
import uuid
import shutil
from fastapi import UploadFile, HTTPException

from app.core.settings import settings


# =========================================================
# ✅ 추가: 업로드 폴더 자동 생성 함수
# =========================================================
def ensure_upload_dir():
    os.makedirs(settings.TODO_UPLOAD_DIR, exist_ok=True)


# =========================================================
# ✅ 추가: 허용 확장자 검사 함수
# =========================================================
def validate_image_file(upload_file: UploadFile):
    allowed_extensions = [".jpg", ".jpeg", ".png", ".gif", ".webp"]

    _, ext = os.path.splitext(upload_file.filename.lower())

    if ext not in allowed_extensions:
        raise HTTPException(
            status_code=400,
            detail="이미지 파일은 jpg, jpeg, png, gif, webp 형식만 업로드할 수 있습니다."
        )


# =========================================================
# ✅ 추가: 파일 저장 함수
# - 실제 서버 폴더에 파일 저장
# - DB에는 상대 경로를 저장
# =========================================================
def save_uploaded_image(upload_file: UploadFile) -> str:
    ensure_upload_dir()
    validate_image_file(upload_file)

    _, ext = os.path.splitext(upload_file.filename.lower())
    unique_filename = f"{uuid.uuid4().hex}{ext}"

    save_path = os.path.join(settings.TODO_UPLOAD_DIR, unique_filename)

    with open(save_path, "wb") as buffer:
        shutil.copyfileobj(upload_file.file, buffer)

    # ✅ DB에는 상대경로 저장
    return f"/media/todos/{unique_filename}"


# =========================================================
# ✅ 추가: 기존 파일 삭제 함수
# - 수정 시 새 파일로 교체하거나 삭제할 때 사용
# =========================================================
def delete_uploaded_file(file_url: str | None):
    if not file_url:
        return

    # 예: /media/todos/abc.png -> media/todos/abc.png
    local_path = file_url.lstrip("/")

    if os.path.exists(local_path):
        os.remove(local_path)
```

from fastapi import (

    HTTPException,
    status,

    Depends,
    BackgroundTasks,
    UploadFile,   # ✅ 추가

from typing import Optional

import asyncio

from app.schemas.todo import Todo
from app.models.todo import TodoDB
from app.models.user import UserDB
from app.db.session import get_db
from app.routes.user import get_current_user
from app.tasks.todo_tasks import write_todo_log, fake_send_notification

# ✅ 추가: 파일 저장 유틸 import
from app.utils.file_handler import save_uploaded_image, delete_uploaded_file

router = APIRouter(prefix="/todos", tags=["Todos"])


# =========================================================
# async / await 연습용 비동기 API
# =========================================================
@router.get("/async-preview")
async def async_preview():
    await asyncio.sleep(2)
    return {"message": "2초 비동기 대기 후 응답이 완료되었습니다."}


# =========================================================
# 내 Todo 목록 조회
# =========================================================
@router.get("", response_model=list[Todo], status_code=status.HTTP_200_OK)
def get_todos(
    keyword: Optional[str] = Query(None, min_length=1, max_length=100, description="제목 검색"),
    done: Optional[bool] = Query(None, description="완료 여부"),
    sort: Optional[str] = Query(None, description="asc 또는 desc"),
    db: Session = Depends(get_db),
    current_user: UserDB = Depends(get_current_user),
):
    query = db.query(TodoDB).filter(TodoDB.user_id == current_user.id)

    if keyword is not None:
        query = query.filter(TodoDB.title.ilike(f"%{keyword}%"))

    if done is not None:
        query = query.filter(TodoDB.done == done)

    if sort is not None and sort not in ["asc", "desc"]:
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST,
            detail="sort 값은 asc 또는 desc만 사용할 수 있습니다."
        )

    if sort == "asc":
        query = query.order_by(TodoDB.id.asc())
    elif sort == "desc":
        query = query.order_by(TodoDB.id.desc())
    else:
        query = query.order_by(TodoDB.id.asc())

    return query.all()


# =========================================================
# 내 Todo 상세 조회
# =========================================================
@router.get("/{todo_id}", response_model=Todo)
def get_todo(

    db: Session = Depends(get_db),
    current_user: UserDB = Depends(get_current_user),
):
    todo = (
        db.query(TodoDB)
        .filter(TodoDB.id == todo_id, TodoDB.user_id == current_user.id)
        .first()
    )

    if not todo:
        raise HTTPException(status_code=404, detail="해당 할 일이 없습니다.")

    return todo


# =========================================================
# ✅ 수정: 내 Todo 생성 + 이미지 업로드
# - form-data 방식 사용
# - title, done, image를 함께 전송 가능
# =========================================================
@router.post("", response_model=Todo, status_code=201)
def create_todo(
    background_tasks: BackgroundTasks,
    title: str = Form(...),                    # ✅ 수정
    done: bool = Form(False),                 # ✅ 수정
    image: UploadFile | None = File(None),    # ✅ 추가
    db: Session = Depends(get_db),
    current_user: UserDB = Depends(get_current_user),
):
    existing_todo = (
        db.query(TodoDB)
        .filter(
            TodoDB.title == title,
            TodoDB.user_id == current_user.id
        )
        .first()
    )

    if existing_todo:
        raise HTTPException(
            status_code=400,
            detail="같은 제목의 할 일이 이미 존재합니다."
        )

    image_path = None

    # ✅ 추가: 이미지가 있으면 저장
    if image is not None and image.filename:
        image_path = save_uploaded_image(image)

    new_todo = TodoDB(
        title=title,
        done=done,
        user_id=current_user.id,
        image_path=image_path  # ✅ 추가
    )

    db.add(new_todo)
    db.commit()
    db.refresh(new_todo)

    background_tasks.add_task(
        write_todo_log,
        "CREATE",
        current_user.username,
        new_todo.title
    )

    background_tasks.add_task(
        fake_send_notification,
        current_user.username,
        f"'{new_todo.title}' Todo가 생성되었습니다."
    )

    return new_todo


# =========================================================
# ✅ 수정: 내 Todo 수정 + 이미지 교체 가능
# - title / done / image 중 일부만 보내도 됨
# =========================================================
@router.put("/{todo_id}", response_model=Todo)
def update_todo(
    todo_id: int,
    background_tasks: BackgroundTasks,
    title: Optional[str] = Form(None),                # ✅ 수정
    done: Optional[bool] = Form(None),                # ✅ 수정
    image: UploadFile | None = File(None),            # ✅ 추가
    db: Session = Depends(get_db),
    current_user: UserDB = Depends(get_current_user),
):
    todo = (
        db.query(TodoDB)
        .filter(TodoDB.id == todo_id, TodoDB.user_id == current_user.id)
        .first()
    )

    if not todo:
        raise HTTPException(status_code=404, detail="수정할 할 일이 없습니다.")

    if title is None and done is None and image is None:
        raise HTTPException(
            status_code=400,
            detail="수정할 데이터를 하나 이상 보내주세요."
        )

    # ✅ 추가: 제목 중복 검사
    if title is not None:
        duplicate_todo = (
            db.query(TodoDB)
            .filter(
                TodoDB.title == title,
                TodoDB.user_id == current_user.id,
                TodoDB.id != todo_id
            )
            .first()
        )
        if duplicate_todo:
            raise HTTPException(
                status_code=400,
                detail="같은 제목의 다른 할 일이 이미 존재합니다."
            )
        todo.title = title

    # ✅ 추가: done 수정
    if done is not None:
        todo.done = done

    # ✅ 추가: 새 이미지 업로드 시 기존 파일 삭제 후 교체
    if image is not None and image.filename:
        delete_uploaded_file(todo.image_path)
        todo.image_path = save_uploaded_image(image)

    db.commit()
    db.refresh(todo)

    background_tasks.add_task(
        write_todo_log,
        "UPDATE",
        current_user.username,
        todo.title
    )

    return todo


# =========================================================
# ✅ 추가: Todo 이미지 삭제 전용 API
# - Todo는 유지하고 이미지 파일만 제거
# =========================================================
@router.delete("/{todo_id}/image", status_code=200)
def delete_todo_image(
    todo_id: int,
    background_tasks: BackgroundTasks,
    db: Session = Depends(get_db),
    current_user: UserDB = Depends(get_current_user),
):
    todo = (
        db.query(TodoDB)
        .filter(TodoDB.id == todo_id, TodoDB.user_id == current_user.id)
        .first()
    )

    if not todo:
        raise HTTPException(status_code=404, detail="해당 할 일이 없습니다.")

    if not todo.image_path:
        raise HTTPException(status_code=400, detail="삭제할 이미지가 없습니다.")

    delete_uploaded_file(todo.image_path)
    todo.image_path = None

    db.commit()
    db.refresh(todo)

    background_tasks.add_task(
        write_todo_log,
        "DELETE_IMAGE",
        current_user.username,
        todo.title
    )

    return {
        "message": "이미지 삭제 완료",
        "todo_id": todo.id
    }


# =========================================================
# 내 Todo 삭제
# - ✅ 수정: Todo 삭제 시 연결된 이미지 파일도 함께 삭제
# =========================================================
@router.delete("/{todo_id}", status_code=200)
def delete_todo(
    todo_id: int,
    background_tasks: BackgroundTasks,
    db: Session = Depends(get_db),
    current_user: UserDB = Depends(get_current_user),
):
    todo = (
        db.query(TodoDB)
        .filter(TodoDB.id == todo_id, TodoDB.user_id == current_user.id)
        .first()
    )

    if not todo:
        raise HTTPException(status_code=404, detail="삭제할 할 일이 없습니다.")

    deleted_title = todo.title

    deleted_data = {
        "id": todo.id,
        "title": todo.title,
        "done": todo.done,
        "user_id": todo.user_id,
        "image_path": todo.image_path  # ✅ 추가
    }

    # ✅ 추가: 연결된 이미지 파일 삭제
    delete_uploaded_file(todo.image_path)

    db.delete(todo)
    db.commit()

    background_tasks.add_task(
        write_todo_log,
        "DELETE",
        current_user.username,
        deleted_title
    )

    return {
        "message": "삭제 완료",
        "deleted": deleted_data
    }

async function createTodo() {

    const title = document.getElementById("create-title").value;
    const imageFile = document.getElementById("create-image").files[0];

    if (!title) {
        showMessage("할 일을 입력하세요", true);
        return;
    }

    const formData = new FormData();
    formData.append("title", title);
    formData.append("done", "false");

    if (imageFile) {
        formData.append("image", imageFile);
    }

    const res = await fetch("/todos", {
        method: "POST",
        headers: {
            "Authorization": "Bearer " + token
        },
        body: formData
    });

    const data = await res.json();

    if (!res.ok) {
        showMessage(data.detail || "생성 실패", true);
        return;
    }

    showMessage("Todo 생성 완료");

    document.getElementById("create-title").value = "";
    document.getElementById("create-image").value = "";

    loadTodos();
}

            box.innerHTML = `<p class="${isError ? 'error' : 'success'}">${message}</p>`;
        }

        function setLoggedIn(username) {
            document.getElementById("login-form").style.display = "none";
            document.getElementById("user-info").style.display = "flex";
            document.getElementById("welcome-text").innerText = `${username}님 로그인`;
        }

        function setLoggedOut() {
            document.getElementById("login-form").style.display = "flex";
            document.getElementById("user-info").style.display = "none";
        }

        function logout(showMsg = true) {
            localStorage.removeItem("token");
            token = null;
            setLoggedOut();
            document.getElementById("todo-list").innerHTML = "";

            if (showMsg) {
                showMessage("로그아웃 되었습니다.");
            }
        }

        async function signup() {
            const username = document.getElementById("login-username").value.trim();
            const password = document.getElementById("login-password").value.trim();

            if (!username || !password) {
                showMessage("아이디와 비밀번호를 입력하세요.", true);
                return;
            }

            const res = await fetch("/users/signup", {
                method: "POST",
                headers: {
                    "Content-Type": "application/json"
                },
                body: JSON.stringify({
                    username: username,
                    password: password
                })
            });

            const data = await res.json();

            if (!res.ok) {
                showMessage(data.detail || "회원가입 실패", true);
                return;
            }

            showMessage("회원가입 성공");
        }

        async function login() {
            const username = document.getElementById("login-username").value.trim();
            const password = document.getElementById("login-password").value.trim();

            if (!username || !password) {
                showMessage("아이디와 비밀번호를 입력하세요.", true);
                return;
            }

            const res = await fetch("/users/login", {
                method: "POST",
                headers: {
                    "Content-Type": "application/json"
                },
                body: JSON.stringify({
                    username: username,
                    password: password
                })
            });

            const data = await res.json();

            if (!res.ok) {
                showMessage(data.detail || "로그인 실패", true);
                return;
            }

            token = data.access_token;
            localStorage.setItem("token", token);

            setLoggedIn(username);
            showMessage("로그인 성공");

            await loadTodos();
        }

        async function loadTodos() {
            if (!token) {
                return;
            }

            const res = await fetch("/todos", {
                headers: {
                    "Authorization": "Bearer " + token
                }
            });

            if (res.status === 401) {
                logout(false);
                showMessage("로그인이 만료되었습니다. 다시 로그인해주세요.", true);
                return;
            }

            const data = await res.json();

            if (!res.ok) {
                showMessage(data.detail || "목록 조회 실패", true);
                return;
            }

            const todos = data;
            const list = document.getElementById("todo-list");
            list.innerHTML = "";

            todos.forEach(todo => {
                const li = document.createElement("li");
                li.className = "todo-item";

                li.innerHTML = `
                    <div class="todo-left">
                        ${todo.image_path ? `<img src="${todo.image_path}" alt="todo image" class="todo-thumb">` : ""}
                        <span class="${todo.done ? 'completed' : ''}">
                            ${todo.title}
                        </span>
                    </div>

                    <div class="todo-buttons">
                        <button class="edit-btn" onclick="toggleDone(${todo.id}, ${todo.done})">완료</button>
                        <button class="delete-btn" onclick="deleteTodo(${todo.id})">삭제</button>
                    </div>
                `;

                list.appendChild(li);
            });
        }

        async function createTodo() {
            if (!token) {
                showMessage("먼저 로그인해주세요.", true);
                return;
            }

            const title = document.getElementById("create-title").value.trim();
            const imageFile = document.getElementById("create-image").files[0];

            if (!title) {
                showMessage("할 일을 입력하세요", true);
                return;
            }

            const formData = new FormData();
            formData.append("title", title);
            formData.append("done", "false");

            if (imageFile) {
                formData.append("image", imageFile);
            }

            const res = await fetch("/todos", {
                method: "POST",
                headers: {
                    "Authorization": "Bearer " + token
                },
                body: formData
            });

            if (res.status === 401) {
                logout(false);
                showMessage("로그인이 만료되었습니다. 다시 로그인해주세요.", true);
                return;
            }

            const data = await res.json();

            if (!res.ok) {
                showMessage(data.detail || "생성 실패", true);
                return;
            }

            showMessage("Todo 생성 완료");
            document.getElementById("create-title").value = "";
            document.getElementById("create-image").value = "";

            await loadTodos();
        }

        async function deleteTodo(id) {
            if (!token) {
                showMessage("먼저 로그인해주세요.", true);
                return;
            }

            const res = await fetch(`/todos/${id}`, {
                method: "DELETE",
                headers: {
                    "Authorization": "Bearer " + token
                }
            });

            if (res.status === 401) {
                logout(false);
                showMessage("로그인이 만료되었습니다. 다시 로그인해주세요.", true);
                return;
            }

            if (!res.ok) {
                const data = await res.json().catch(() => ({}));
                showMessage(data.detail || "삭제 실패", true);
                return;
            }

            showMessage("삭제 완료");
            await loadTodos();
        }

        async function toggleDone(id, done) {
            if (!token) {
                showMessage("먼저 로그인해주세요.", true);
                return;
            }

            const formData = new FormData();
            formData.append("done", (!done).toString());

            const res = await fetch(`/todos/${id}`, {
                method: "PUT",
                headers: {
                    "Authorization": "Bearer " + token
                },
                body: formData
            });

            if (res.status === 401) {
                logout(false);
                showMessage("로그인이 만료되었습니다. 다시 로그인해주세요.", true);
                return;
            }

            const data = await res.json().catch(() => ({}));

            if (!res.ok) {
                showMessage(data.detail || "수정 실패", true);
                return;
            }

            showMessage("수정 완료");
            await loadTodos();
        }

        async function checkLogin() {
            if (!token) {
                return;
            }

            const res = await fetch("/users/me", {
                headers: {
                    "Authorization": "Bearer " + token
                }
            });

            if (!res.ok) {
                logout(false);
                return;
            }

            const data = await res.json();
            setLoggedIn(data.username);
            await loadTodos();
        }

        checkLogin();

---
### Swagger에서 테스트
이번 단계에서 가장 중요한 점은  
파일 업로드 API는 JSON이 아니라 form-data 방식이라는 점입니다.

`1.` 회원가입
```
POST /users/signup
```

`2.` 로그인
```
POST /users/login
```
토큰 발급 후 Swagger 우측 상단 **Authorize**에 넣습니다.

Bearer 발급받은토큰

`3.` Todo + 이미지 생성
```
POST /todos
```
Swagger에서 아래처럼 입력하시면 됩니다.

- `title`: `강아지 사진 정리`
- `done`: `false`
- `image`: 파일 선택
    

즉, body를 JSON으로 넣는 것이 아니라  
Swagger가 자동으로 파일 선택창을 보여줍니다.

`4.` Todo 수정 + 이미지 교체
```
PUT /todos/{todo_id}
```

- `title`만 수정 가능
- `done`만 수정 가능
- `image`만 새로 업로드 가능
- 셋 중 여러 개 같이 보내도 됩니다
    

`5.` 이미지 삭제
```
DELETE /todos/{todo_id}/image
```

이 API는 Todo는 남기고 이미지 파일만 지웁니다.

`6.` Todo 삭제
```
DELETE /todos/{todo_id}
```
Todo 삭제 시 연결된 이미지 파일도 같이 삭제됩니다.

### 이 코드가 동작하는 흐름
예를 들어 사용자가 이미지를 포함해서 Todo를 생성하면 흐름은 아래와 같습니다.
```
사용자 요청  
→ /todos POST  
→ title, done, image 수신  
→ save_uploaded_image(image) 실행  
→ media/todos/폴더에 실제 파일 저장  
→ DB의 image_path 컬럼에 "/media/todos/파일명.png" 저장  
→ Todo 응답 반환
```
브라우저에서 나중에 이 경로를 열면 실제 이미지가 보입니다.

예:
http://127.0.0.1:8000/media/todos/abc123.png

---

### 꼭 이해하셔야 하는 점

`1.` DB에는 파일 자체를 저장하지 않습니다
이번 구조는 이미지 파일은 폴더에 저장하고,  
DB에는 파일 경로 문자열만 저장합니다.

예:
- 실제 파일: `media/todos/abc123.png`
- DB 저장값: `/media/todos/abc123.png`
    
이 방식이 가장 단순하고 초보자가 연습하기 좋습니다.

`2.` 파일 업로드가 들어가면 JSON 방식과 다릅니다
기존에 `TodoCreate`처럼 JSON body를 받던 방식과 달리,  
파일 업로드가 들어가면 `Form(...)`, `File(...)` 조합으로 받아야 합니다.

`3.` 수정 시 기존 이미지 삭제 후 새 이미지 저장
파일이 계속 쌓이지 않도록  
이미지 교체 시 기존 파일을 먼저 삭제하도록 넣어두었습니다.
