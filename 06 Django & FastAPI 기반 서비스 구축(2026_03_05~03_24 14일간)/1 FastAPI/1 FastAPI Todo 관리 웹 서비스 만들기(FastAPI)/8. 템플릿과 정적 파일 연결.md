```python
from fastapi import FastAPI, Request
from fastapi.responses import JSONResponse
from fastapi.exceptions import RequestValidationError
from fastapi.middleware.cors import CORSMiddleware
from fastapi.staticfiles import StaticFiles   # ✅ 추가: static 파일 연결
from fastapi.templating import Jinja2Templates  # ✅ 추가: Jinja2 템플릿
from app.routes.item import router as item_router

app = FastAPI(
    title="FastAPI Todo API",
    description="Pydantic 검증과 Path / Query Parameter가 포함된 Todo CRUD API",
    version="1.4.0"  # ✅ 수정: 버전 업데이트
)

# CORS 설정
app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)

# 라우터 등록
app.include_router(item_router)


# ✅ 추가: static 폴더 연결
app.mount("/static", StaticFiles(directory="static"), name="static")


# ✅ 추가: 템플릿 폴더 설정
templates = Jinja2Templates(directory="app/templates")


# ✅ 추가: HTML 페이지 렌더링
@app.get("/")
def todo_page(request: Request):
    return templates.TemplateResponse(
        "index.html",
        {"request": request}
    )
```

`app/templates/index.html`
```html
<!DOCTYPE html>
<html lang="ko">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">

    <!-- 브라우저 탭 제목 -->
    <title>FastAPI Todo</title>

    <!-- static 폴더에 있는 CSS 파일 연결 -->
    <link rel="stylesheet" href="/static/style.css">
</head>
<body>
    <div class="todo-container">
        <!-- 페이지 메인 제목 -->
        <h1>Todo List</h1>

        <!-- =========================
             Todo 생성 영역
             - 사용자가 새 할 일 제목을 입력
             - 추가 버튼 클릭 시 createTodo() 실행
        ========================== -->
        <div class="todo-input-box">
            <input type="text" id="create-title" placeholder="할 일을 입력하세요">
            <button onclick="createTodo()">추가</button>
        </div>

        <!-- =========================
             검색 / 필터 / 정렬 영역
             - keyword: 제목 검색
             - done: 완료/미완료 필터
             - sort: 정렬 방식 선택
             - 검색 버튼: searchTodos() 실행
             - 초기화 버튼: resetSearch() 실행
        ========================== -->
        <div class="search-filter-box">
            <input type="text" id="search-keyword" placeholder="검색어를 입력하세요">

            <select id="filter-done">
                <option value="">전체 상태</option>
                <option value="true">완료</option>
                <option value="false">미완료</option>
            </select>

            <select id="sort-order">
                <option value="">기본 정렬</option>
                <option value="asc">오름차순</option>
                <option value="desc">내림차순</option>
            </select>

            <button class="search-btn" onclick="searchTodos()">검색</button>
            <button class="reset-btn" onclick="resetSearch()">초기화</button>
        </div>

        <!-- 성공 / 실패 메시지를 보여주는 영역 -->
        <div id="message-box"></div>

        <!-- Todo 목록이 동적으로 렌더링되는 영역 -->
        <ul id="todo-list" class="todo-list"></ul>
    </div>

    <script>
        // =========================
        // 공통 메시지 출력 함수
        // - 성공/실패 메시지를 화면에 보여줍니다.
        // - isError=true이면 에러 스타일로 출력합니다.
        // =========================
        function showMessage(message, isError = false) {
            const box = document.getElementById("message-box");
            box.innerHTML = `<p class="${isError ? 'error' : 'success'}">${message}</p>`;
        }

        // =========================
        // 검색 조건을 Query String으로 만드는 함수
        // - 입력된 검색어, 완료 여부, 정렬값을 읽습니다.
        // - 값이 있는 것만 URL 파라미터에 붙입니다.
        // - 예:
        //   /todos?keyword=공부&done=false&sort=desc
        // =========================
        function buildQueryString() {
            const keyword = document.getElementById("search-keyword").value.trim();
            const done = document.getElementById("filter-done").value;
            const sort = document.getElementById("sort-order").value;

            const params = new URLSearchParams();

            if (keyword) {
                params.append("keyword", keyword);
            }

            if (done !== "") {
                params.append("done", done);
            }

            if (sort) {
                params.append("sort", sort);
            }

            const queryString = params.toString();
            return queryString ? `/todos?${queryString}` : "/todos";
        }

        // =========================
        // Todo 목록 조회 함수
        // - 기본값은 /todos 전체 조회입니다.
        // - 검색/필터/정렬 URL이 들어오면 그 조건대로 조회합니다.
        // - 서버 응답(JSON 배열)을 받아 목록에 렌더링합니다.
        // - 데이터가 없으면 "조회된 Todo가 없습니다." 메시지를 보여줍니다.
        // =========================
        async function loadTodos(url = "/todos") {
            const res = await fetch(url);
            const todos = await res.json();

            const list = document.getElementById("todo-list");
            list.innerHTML = "";

            if (!Array.isArray(todos) || todos.length === 0) {
                list.innerHTML = `<li class="empty-message">조회된 Todo가 없습니다.</li>`;
                return;
            }

            todos.forEach(todo => {
                const li = document.createElement("li");
                li.className = "todo-item";

                // 제목에 작은따옴표가 있을 때 onclick 문자열이 깨지지 않도록 처리
                const safeTitle = todo.title.replace(/'/g, "\\'");

                li.innerHTML = `
                    <div class="todo-left">
                        <input
                            type="checkbox"
                            ${todo.done ? "checked" : ""}
                            onchange="toggleDone(${todo.id}, this.checked)"
                        >
                        <span class="todo-text ${todo.done ? 'completed' : ''}" id="text-${todo.id}">
                            ${todo.title}
                        </span>
                    </div>

                    <div class="todo-buttons">
                        <button class="edit-btn" onclick="editTodo(${todo.id}, '${safeTitle}', ${todo.done})">수정</button>
                        <button class="delete-btn" onclick="deleteTodo(${todo.id})">삭제</button>
                    </div>
                `;

                list.appendChild(li);
            });
        }

        // =========================
        // Todo 생성 함수
        // - 입력창의 제목 값을 읽습니다.
        // - 비어 있으면 생성하지 않고 에러 메시지를 보여줍니다.
        // - /todos로 POST 요청을 보내 새 Todo를 생성합니다.
        // - 성공하면 입력창을 비우고 목록을 다시 조회합니다.
        // =========================
        async function createTodo() {
            const titleInput = document.getElementById("create-title");
            const title = titleInput.value.trim();

            if (!title) {
                showMessage("할 일을 입력해주세요.", true);
                return;
            }

            const res = await fetch("/todos", {
                method: "POST",
                headers: {
                    "Content-Type": "application/json"
                },
                body: JSON.stringify({
                    title: title,
                    done: false
                })
            });

            const data = await res.json();

            if (!res.ok) {
                showMessage(data.detail || data.message || "생성 실패", true);
                return;
            }

            titleInput.value = "";
            showMessage("Todo가 추가되었습니다.");

            searchTodos();
        }

        // =========================
        // Todo 삭제 함수
        // - 전달받은 todoId를 이용해서 /todos/{id}로 DELETE 요청을 보냅니다.
        // - 성공하면 목록을 다시 조회합니다.
        // =========================
        async function deleteTodo(todoId) {
            const res = await fetch(`/todos/${todoId}`, {
                method: "DELETE"
            });

            const data = await res.json();

            if (!res.ok) {
                showMessage(data.detail || data.message || "삭제 실패", true);
                return;
            }

            showMessage("Todo가 삭제되었습니다.");
            searchTodos();
        }

        // =========================
        // Todo 완료 상태 변경 함수
        // - 체크박스 상태(checked)를 받아 완료 여부를 수정합니다.
        // - 현재 화면에 보이는 제목(text)도 함께 보내서 PUT 요청을 보냅니다.
        // - 성공하면 목록을 다시 조회합니다.
        // =========================
        async function toggleDone(todoId, checked) {
            const text = document.getElementById(`text-${todoId}`).innerText;

            const res = await fetch(`/todos/${todoId}`, {
                method: "PUT",
                headers: {
                    "Content-Type": "application/json"
                },
                body: JSON.stringify({
                    title: text,
                    done: checked
                })
            });

            const data = await res.json();

            if (!res.ok) {
                showMessage(data.detail || data.message || "상태 변경 실패", true);
                return;
            }

            searchTodos();
        }

        // =========================
        // Todo 제목 수정 함수
        // - prompt 창으로 기존 제목을 보여주고 새 제목을 입력받습니다.
        // - 사용자가 취소하면 아무 작업도 하지 않습니다.
        // - 빈 문자열이면 수정하지 않고 에러 메시지를 보여줍니다.
        // - /todos/{id}로 PUT 요청을 보내 제목을 수정합니다.
        // - done 값은 기존 상태(oldDone)를 그대로 유지합니다.
        // =========================
        async function editTodo(todoId, oldTitle, oldDone) {
            const newTitle = prompt("수정할 제목을 입력하세요.", oldTitle);

            if (newTitle === null) return;

            const trimmed = newTitle.trim();

            if (!trimmed) {
                showMessage("제목은 비워둘 수 없습니다.", true);
                return;
            }

            const res = await fetch(`/todos/${todoId}`, {
                method: "PUT",
                headers: {
                    "Content-Type": "application/json"
                },
                body: JSON.stringify({
                    title: trimmed,
                    done: oldDone
                })
            });

            const data = await res.json();

            if (!res.ok) {
                showMessage(data.detail || data.message || "수정 실패", true);
                return;
            }

            showMessage("Todo가 수정되었습니다.");
            searchTodos();
        }

        // =========================
        // 검색 실행 함수
        // - 현재 입력된 검색/필터/정렬 조건을 URL로 만듭니다.
        // - 그 URL로 Todo 목록을 다시 조회합니다.
        // =========================
        function searchTodos() {
            const url = buildQueryString();
            loadTodos(url);
        }

        // =========================
        // 검색 조건 초기화 함수
        // - 검색어, 완료 여부, 정렬 값을 모두 초기화합니다.
        // - 전체 Todo 목록을 다시 불러옵니다.
        // =========================
        function resetSearch() {
            document.getElementById("search-keyword").value = "";
            document.getElementById("filter-done").value = "";
            document.getElementById("sort-order").value = "";

            loadTodos();
            showMessage("검색 조건이 초기화되었습니다.");
        }

        // =========================
        // 검색어 입력창 엔터키 이벤트
        // - 사용자가 검색어 입력 후 Enter를 누르면
        //   검색 버튼을 누른 것처럼 searchTodos()를 실행합니다.
        // =========================
        document.getElementById("search-keyword").addEventListener("keydown", function(event) {
            if (event.key === "Enter") {
                searchTodos();
            }
        });

        // =========================
        // 페이지 최초 실행
        // - 화면이 열리면 전체 Todo 목록을 바로 조회합니다.
        // =========================
        loadTodos();
    </script>
</body>
</html>
```
