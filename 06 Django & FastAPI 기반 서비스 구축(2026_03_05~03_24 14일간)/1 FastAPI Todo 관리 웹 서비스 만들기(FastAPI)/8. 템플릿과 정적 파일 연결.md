`app/main.py`
```python
from fastapi import FastAPI, Request
from fastapi.responses import JSONResponse
from fastapi.exceptions import RequestValidationError
from fastapi.middleware.cors import CORSMiddleware
from fastapi.staticfiles import StaticFiles   # ✅ 추가: static 파일 연결
from fastapi.templating import Jinja2Templates  # ✅ 추가: Jinja2 템플릿
from app.routes.item import router as item_router

app = FastAPI(
    title="FastAPI Todo API",
    description="Pydantic 검증과 Path / Query Parameter가 포함된 Todo CRUD API",
    version="1.4.0"  # ✅ 수정: 버전 업데이트
)

# CORS 설정
app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)

# 라우터 등록
app.include_router(item_router)


# ✅ 추가: static 폴더 연결
app.mount("/static", StaticFiles(directory="static"), name="static")


# ✅ 추가: 템플릿 폴더 설정
templates = Jinja2Templates(directory="app/templates")


# ✅ 추가: HTML 페이지 렌더링
@app.get("/")
def todo_page(request: Request):
    return templates.TemplateResponse(
        "index.html",
        {"request": request}
    )

# ✅ 삭제: HTML 페이지 렌더링을 위해

@app.exception_handler(RequestValidationError)
async def validation_exception_handler(request: Request, exc: RequestValidationError):
    return JSONResponse(
        status_code=422,
        content={
            "message": "입력값 검증에 실패했습니다.",
            "errors": exc.errors()
        }
    )
```

`app/templates/index.html`
```html
<!DOCTYPE html>
<html lang="ko">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>FastAPI Todo</title>
    <link rel="stylesheet" href="/static/style.css">
</head>
<body>
    <div class="todo-container">
        <h1>Todo List</h1>

        <!-- 생성 영역 -->
        <div class="todo-input-box">
            <input type="text" id="create-title" placeholder="할 일을 입력하세요">
            <button onclick="createTodo()">추가</button>
        </div>

        <!-- 메시지 -->
        <div id="message-box"></div>

        <!-- 목록 -->
        <ul id="todo-list" class="todo-list"></ul>
    </div>

    <script>
        function showMessage(message, isError = false) {
            const box = document.getElementById("message-box");
            box.innerHTML = `<p class="${isError ? 'error' : 'success'}">${message}</p>`;
        }

        async function loadTodos() {
            const res = await fetch("/todos");
            const todos = await res.json();

            const list = document.getElementById("todo-list");
            list.innerHTML = "";

            todos.forEach(todo => {
                const li = document.createElement("li");
                li.className = "todo-item";

                li.innerHTML = `
                    <div class="todo-left">
                        <input 
                            type="checkbox" 
                            ${todo.done ? "checked" : ""} 
                            onchange="toggleDone(${todo.id}, this.checked)"
                        >
                        <span class="todo-text ${todo.done ? 'completed' : ''}" id="text-${todo.id}">
                            ${todo.title}
                        </span>
                    </div>

                    <div class="todo-buttons">
                        <button class="edit-btn" onclick="editTodo(${todo.id}, '${todo.title.replace(/'/g, "\\'")}', ${todo.done})">수정</button>
                        <button class="delete-btn" onclick="deleteTodo(${todo.id})">삭제</button>
                    </div>
                `;

                list.appendChild(li);
            });
        }

        async function createTodo() {
            const titleInput = document.getElementById("create-title");
            const title = titleInput.value.trim();

            if (!title) {
                showMessage("할 일을 입력해주세요.", true);
                return;
            }

            const res = await fetch("/todos", {
                method: "POST",
                headers: {
                    "Content-Type": "application/json"
                },
                body: JSON.stringify({
                    title: title,
                    done: false
                })
            });

            const data = await res.json();

            if (!res.ok) {
                showMessage(data.detail || "생성 실패", true);
                return;
            }

            titleInput.value = "";
            showMessage("Todo가 추가되었습니다.");
            loadTodos();
        }

        async function deleteTodo(todoId) {
            const res = await fetch(`/todos/${todoId}`, {
                method: "DELETE"
            });

            const data = await res.json();

            if (!res.ok) {
                showMessage(data.detail || "삭제 실패", true);
                return;
            }

            showMessage("Todo가 삭제되었습니다.");
            loadTodos();
        }

        async function toggleDone(todoId, checked) {
            const text = document.getElementById(`text-${todoId}`).innerText;

            const res = await fetch(`/todos/${todoId}`, {
                method: "PUT",
                headers: {
                    "Content-Type": "application/json"
                },
                body: JSON.stringify({
                    title: text,
                    done: checked
                })
            });

            const data = await res.json();

            if (!res.ok) {
                showMessage(data.detail || "상태 변경 실패", true);
                return;
            }

            loadTodos();
        }

        async function editTodo(todoId, oldTitle, oldDone) {
            const newTitle = prompt("수정할 제목을 입력하세요.", oldTitle);

            if (newTitle === null) return;

            const trimmed = newTitle.trim();
            if (!trimmed) {
                showMessage("제목은 비워둘 수 없습니다.", true);
                return;
            }

            const res = await fetch(`/todos/${todoId}`, {
                method: "PUT",
                headers: {
                    "Content-Type": "application/json"
                },
                body: JSON.stringify({
                    title: trimmed,
                    done: oldDone
                })
            });

            const data = await res.json();

            if (!res.ok) {
                showMessage(data.detail || "수정 실패", true);
                return;
            }

            showMessage("Todo가 수정되었습니다.");
            loadTodos();
        }

        loadTodos();
    </script>
</body>
</html>
```
