`database.py`파일을 `db/base.py`, `db/session.py`, `db/init_db.py`로 분리하는것이 핵심

파일 및 폴더 생성
```bash
mkdir -p app/db
touch app/db/__init__.py  
touch app/db/base.py
touch app/db/session.py 
touch app/db/init_db.py

mkdir -p app/core
touch app/core/__init__.py  
touch app/core/settings.py  
```

### 왜 이걸 해야 하나?
핵심은 두 가지입니다.

`database.py`파일에서
`(1) DB session 관리`
	라우터마다 DB 세션을 안정적으로 열고 닫기 위해 필요해요.

즉:
```python
db: Session = Depends(get_db)
```
이 구조를 계속 쓸 건데, 이 `get_db()`를 깔끔하게 따로 관리하는 게 좋습니다.

`(2) settings 파일 분리`
	지금은 DB 주소가 코드에 박혀 있죠.
```python
SQLALCHEMY_DATABASE_URL = "sqlite:///./todo.db"
```

이걸 settings로 빼두면 나중에
- SQLite
- PostgreSQL
- Docker 환경
- 운영/개발 환경으로 바꿀 때 훨씬 쉬워져요.

### DRF 요청 흐름 (익숙한 구조)
```
Client
   ↓
URL
   ↓
View
   ↓
Serializer
   ↓
Model
   ↓
DB
```

예를 들어 Todo 조회
```
Client
 ↓
urls.py
 ↓
TodoViewSet
 ↓
TodoSerializer
 ↓
Todo model
 ↓
PostgreSQL
```
즉 DRF는 5단계 구조입니다.

### 비교 FastAPI 요청 흐름 (같은 구조)
FastAPI도 사실 동일합니다.
```
Client
 ↓
Router
 ↓
Schema
 ↓
Model
 ↓
DB Session
 ↓
DB
```
DRF와 1:1 대응하면 이렇게 됩니다.
![[Pasted image 20260314160148.png]]

위에 피그마로 만들어놓은 이미지 기준으로 다시 그리면 이렇게 됩니다.
```
Client
   ↓
routes/item.py
   ↓
schemas/todo.py
   ↓
models/todo.py
   ↓
db/session.py
   ↓
SQLite
```
즉 DRF와 동일한 흐름입니다.

그러면 `main.py` 역할 (DRF 기준)은 아래 3개 파일이 합쳐진 역할입니다.
```
manage.py + urls.py + settings.py
```

지금 구조를 DRF 스타일로 다시 그리면
```
app
 ├─ core
 │   └─ settings.py   ← Django settings
 │
 ├─ db
 │   ├─ session.py    ← Django ORM connection
 │   ├─ base.py       ← models.Model
 │   └─ init_db.py    ← migrate
 │
 ├─ models
 │   └─ todo.py       ← models.py
 │
 ├─ schemas
 │   └─ todo.py       ← serializers.py
 │
 ├─ routes
 │   └─ item.py       ← views.py
 │
 └─ main.py           ← manage.py + urls.py
```

```python
from app.core.settings import settings

engine = create_engine(
    settings.DATABASE_URL,
    connect_args={"check_same_thread": False}  # SQLite 전용 옵션
)

# ✅ 추가: 세션 팩토리 생성
SessionLocal = sessionmaker(
    autocommit=False,
    autoflush=False,
    bind=engine
)


# ✅ 추가: FastAPI Depends에서 사용할 DB 세션 함수
def get_db():
    db = SessionLocal()
    try:
        yield db
    finally:
        db.close()
```

`app/routes/item.py` : urls + views의 역할
```python
from fastapi import APIRouter, HTTPException, status, Query, Depends
from typing import Optional
from fastapi import Header, Cookie, Response
from sqlalchemy.orm import Session

from app.schemas.todo import Todo, TodoCreate, TodoUpdate
from app.models.todo import TodoDB


# 기존 database를 session으로 변경
# from app.database import get_db

# ✅ 변경: get_db import 위치 변경
from app.db.session import get_db


router = APIRouter()

# 전체 조회 + 검색/필터
@router.get(
    "/todos",
    response_model=list[Todo],
    status_code=status.HTTP_200_OK,
    responses={
        200: {"description": "전체 Todo 목록 조회 성공"},
        400: {"description": "잘못된 정렬 값 요청"}
    }
)
def get_todos(
    keyword: Optional[str] = Query(None, min_length=1, max_length=100, description="제목 검색"),
    done: Optional[bool] = Query(None, description="완료 여부"),
    sort: Optional[str] = Query(None, description="asc 또는 desc"),
    db: Session = Depends(get_db)   # ✅ 유지: 세션 의존성 주입
):
    query = db.query(TodoDB)

    if keyword is not None:
        query = query.filter(TodoDB.title.ilike(f"%{keyword}%"))

    if done is not None:
        query = query.filter(TodoDB.done == done)

    if sort is not None and sort not in ["asc", "desc"]:
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST,
            detail="sort 값은 asc 또는 desc만 사용할 수 있습니다."
        )

    if sort == "asc":
        query = query.order_by(TodoDB.id.asc())
    elif sort == "desc":
        query = query.order_by(TodoDB.id.desc())
    else:
        query = query.order_by(TodoDB.id.asc())

    return query.all()


@router.get("/todos/{todo_id}", response_model=Todo)
def get_todo(todo_id: int, db: Session = Depends(get_db)):
    todo = db.query(TodoDB).filter(TodoDB.id == todo_id).first()

    if not todo:
        raise HTTPException(status_code=404, detail="해당 할 일이 없습니다.")

    return todo


@router.post("/todos", response_model=Todo, status_code=201)
def create_todo(todo_data: TodoCreate, db: Session = Depends(get_db)):
    existing_todo = db.query(TodoDB).filter(TodoDB.title == todo_data.title).first()
    if existing_todo:
        raise HTTPException(status_code=400, detail="같은 제목의 할 일은 이미 존재합니다.")

    new_todo = TodoDB(
        title=todo_data.title,
        done=todo_data.done
    )

    db.add(new_todo)
    db.commit()
    db.refresh(new_todo)

    return new_todo


@router.put("/todos/{todo_id}", response_model=Todo)
def update_todo(todo_id: int, todo_data: TodoUpdate, db: Session = Depends(get_db)):
    update_data = todo_data.model_dump(exclude_unset=True)

    if not update_data:
        raise HTTPException(status_code=400, detail="수정할 데이터를 하나 이상 보내주세요.")

    todo = db.query(TodoDB).filter(TodoDB.id == todo_id).first()
    if not todo:
        raise HTTPException(status_code=404, detail="수정할 할 일이 없습니다.")

    if "title" in update_data:
        duplicate_todo = (
            db.query(TodoDB)
            .filter(TodoDB.title == update_data["title"], TodoDB.id != todo_id)
            .first()
        )
        if duplicate_todo:
            raise HTTPException(status_code=400, detail="같은 제목의 다른 할 일이 이미 존재합니다.")

    for key, value in update_data.items():
        setattr(todo, key, value)

    db.commit()
    db.refresh(todo)

    return todo


@router.delete("/todos/{todo_id}", status_code=200)
def delete_todo(todo_id: int, db: Session = Depends(get_db)):
    todo = db.query(TodoDB).filter(TodoDB.id == todo_id).first()

    if not todo:
        raise HTTPException(status_code=404, detail="삭제할 할 일이 없습니다.")

    deleted_data = {
        "id": todo.id,
        "title": todo.title,
        "done": todo.done
    }

    db.delete(todo)
    db.commit()

    return {
        "message": "삭제 완료",
        "deleted": deleted_data
    }


@router.get("/headers-demo")
def read_headers(
    user_agent: Optional[str] = Header(None),
    x_token: Optional[str] = Header(None)
):
    return {
        "message": "요청 헤더를 읽었습니다.",

    }
```

`app/main.py`
```python
from fastapi import FastAPI, Request
from fastapi.responses import JSONResponse
from fastapi.exceptions import RequestValidationError
from fastapi.middleware.cors import CORSMiddleware
from fastapi.staticfiles import StaticFiles
from fastapi.templating import Jinja2Templates
from app.routes.item import router as item_router

# ✅ 수정: settings import
from app.core.settings import settings

# ✅ 수정: init_db 함수 import
from app.db.init_db import init_db


# ✅ 수정: 앱 시작 전에 DB 초기화
init_db()

app = FastAPI(
    title=settings.PROJECT_NAME,
    description="FastAPI + SQLAlchemy 기반 Todo CRUD API",
    version=settings.VERSION
)

app.add_middleware(
    CORSMiddleware,
    allow_origins=settings.ALLOWED_ORIGINS,
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)

app.include_router(item_router)

app.mount("/static", StaticFiles(directory="static"), name="static")

templates = Jinja2Templates(directory="app/templates")


@app.get("/")
def home():
    return {"message": "FastAPI Todo API 실행 중 (DB session / settings 분리 완료)"}


@app.get("/page")
def todo_page(request: Request):
    return templates.TemplateResponse(
        "index.html",
        {"request": request}
    )


@app.exception_handler(RequestValidationError)
async def validation_exception_handler(request: Request, exc: RequestValidationError):
    return JSONResponse(
        status_code=422,
        content={
            "message": "입력값 검증에 실패했습니다.",
            "errors": exc.errors()
        }
    )
```
